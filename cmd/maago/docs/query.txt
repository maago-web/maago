package main

import (
	"context"
	"log"

	"github.com/maago-web/maago"
	_ "modernc.org/sqlite"

	"example.com/app/auth"
	"example.com/app/blog"
)

func main() {
	ctx := context.Background()
	db, err := maago.OpenDB(ctx, maago.DBConfig{})
	if err != nil {
		log.Fatal(err)
	}
	auth.SiteURL = "http://localhost:8000"
	h, err := maago.Handler(maago.Config{DB: db, Apps: []maago.App{auth.App, blog.App}, Auth: auth.Current, LoginURL: "/login"})
	if err != nil {
		log.Fatal(err)
	}
	if err := maago.Serve(ctx, ":8000", h); err != nil {
		log.Fatal(err)
	}
}

package blog

import (
	"time"

	"github.com/maago-web/maago"
)

type Post struct {
	maago.Model

	ID          int64
	AuthorID    *int64 `maago:"fk=auth.User"`
	Title       string `maago:"max=200"`
	Status      string `maago:"choices=draft/published,default='draft'"`
	Views       int64  `maago:"default=0"`
	PublishedAt *time.Time
}

type Comment struct {
	maago.Model

	ID     int64
	PostID int64 `maago:"fk=Post"`
	Text   string
}
-- blog/policies.sql --
-- name: visible Post p
-- get: GetPost
-- The posts the user may read: those published by now, and their own.
-- param: now time.Time
p.status = 'published' AND (p.published_at IS NULL OR p.published_at <= :now)
  OR p.author_id = :viewer_id
-- blog/queries/search_posts.sql --
-- name: SearchPosts :many PostSummary
-- param: q?      string
-- param: status any/draft/published
-- param: ids     []int64
-- policy: visible p
SELECT p.id, p.title, p.published_at
FROM blog_post p
WHERE {
  visible(p)
  {q}       AND p.title LIKE '%' || :q || '%'
  {status=draft|status=published} AND p.status = :status
  {ids}     AND p.id IN (:ids)
}
ORDER BY p.published_at DESC
-- blog/queries/post_by_title.sql --
-- name: PostByTitle :one Post
-- param: title string
-- policy: visible blog_post
SELECT * FROM blog_post WHERE title = :title AND visible(blog_post)
-- blog/queries/posts_with_counts.sql --
-- name: PostsWithCounts :many PostCount
-- column: comments int64
-- column: last_comment? string
-- policy: visible p
SELECT p.id, p.title, count(c.id) AS comments, max(c.text) AS last_comment
FROM blog_post p LEFT JOIN blog_comment c ON c.post_id = p.id
WHERE visible(p)
GROUP BY p.id, p.title
-- blog/queries/most_views.sql --
-- name: MostViews :one int64
-- policy: none blog_post a number, which shows no post
SELECT COALESCE(max(views), 0) FROM blog_post
-- blog/queries/count_view.sql --
-- name: CountView :exec
-- param: id int64
UPDATE blog_post SET views = views + 1 WHERE id = :id

package blog

import (
	"context"
	"time"

	"github.com/maago-web/maago"

	"example.com/app/auth"
)

// APIPosts lists the posts u may read: nil is nobody.
func APIPosts(ctx context.Context, u *auth.User, db maago.DB, in struct{}) ([]PostSummary, error) {
	return SearchPosts(ctx, u, db, SearchPostsParams{Status: "any", Now: time.Now()})
}

// Show runs the generated functions.
func Show(ctx context.Context, u *auth.User, tx maago.Tx, id int64) ([]PostSummary, error) {
	now := time.Now()
	if _, err := CountView(ctx, tx, CountViewParams{ID: id}); err != nil {
		return nil, err
	}
	if _, err := MostViews(ctx, tx); err != nil {
		return nil, err
	}
	if _, err := PostsWithCounts(ctx, u, tx, PostsWithCountsParams{Now: now}); err != nil {
		return nil, err
	}
	if _, err := PostByTitle(ctx, u, tx, PostByTitleParams{Title: "Hello", Now: now}); err != nil {
		return nil, err
	}
	if _, err := GetPost(ctx, u, tx, id, now); err != nil {
		return nil, err
	}
	return SearchPosts(ctx, u, tx, SearchPostsParams{Q: maago.Ptr("hello"), Status: "any", IDs: []int64{1, 2}, Now: now})
}

func TestPermissions(t *testing.T) {
	h, err := maago.Handler(maago.Config{DB: maagotest.DB(t), Apps: []maago.App{auth.App, App}, Auth: auth.Current, LoginURL: "/login"})
	if err != nil {
		t.Fatal(err)
	}
	maagotest.Permissions(t, []maago.App{App}, []maagotest.Role{{Name: "nobody", Handler: h}},
		maagotest.Row{Route: "api_posts", URL: APIPostsURL(), Want: map[string]int{"nobody": 200}})
}
