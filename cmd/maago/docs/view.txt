		log.Fatal(err)

	}

		log.Fatal(err)
	}
}

package blog

import (
	"context"
	"fmt"
	"net/http"
	"strconv"

	"github.com/maago-web/maago"

	"example.com/app/auth"
)

type PostListIn struct {
	Page *int64 `maago:"query=page"` // ?page=, nil when missing
}

type PostListPage struct {
	Posts    []Post
	SignedIn bool
}

// PostList is for anybody: u is nil when nobody is signed in.
func PostList(ctx context.Context, u *auth.User, db maago.DB, in PostListIn) (PostListPage, error) {
	posts, err := AllPosts(ctx, db)
	return PostListPage{Posts: posts, SignedIn: u != nil}, err
}

type PostDetailIn struct {
	ID int64 `maago:"path=id"` // {id}; /post/abc is a 404
}

type PostDetailPage struct {
	Post Post
}

func PostDetail(ctx context.Context, db maago.DB, in PostDetailIn) (PostDetailPage, error) {
	p, err := GetPost(ctx, db, in.ID) // sql.ErrNoRows: 404
	return PostDetailPage{Post: p}, err
}

type NewPostIn struct {
	Form PostForm `maago:"form"`
}

type PostFormPage struct {
	Form PostForm
}

// NewPost draws the empty form; u requires a sign-in.
func NewPost(ctx context.Context, u auth.User, in NewPostIn) (PostFormPage, error) {
	return PostFormPage{Form: in.Form}, nil
}

// CreatePost writes, in the request's transaction.
func CreatePost(ctx context.Context, u auth.User, tx maago.Tx, in NewPostIn) (maago.Result[PostFormPage], error) {
	f := in.Form
	if !f.Valid() {
		return maago.Show(PostFormPage{Form: f}), nil
	}
	p, err := InsertPost(ctx, tx, NewPostInput{AuthorID: u.ID, Title: f.Title.Value})
	if err != nil {
		return maago.Result[PostFormPage]{}, err // rolled back
	}
	maago.AfterCommit(ctx, func(ctx context.Context) { fmt.Println("new post", p.ID) })
	return maago.SeeOther[PostFormPage](PostDetailURL(p.ID)), nil
}

// CountView takes maago.DB on a writing route, so the route says maago.NoTx.
func CountView(ctx context.Context, db maago.DB, in PostDetailIn) (maago.Result[PostDetailPage], error) {
	if _, err := CountPostView(ctx, db, CountPostViewParams{ID: in.ID}); err != nil {
		return maago.Result[PostDetailPage]{}, err
	}
	return maago.SeeOther[PostDetailPage](PostDetailURL(in.ID)), nil
}

func APIPosts(ctx context.Context, db maago.DB, in struct{}) ([]Post, error) {
	return AllPosts(ctx, db)
}

// PostText answers plain text, so it is a func(w, r) and reads what it
// needs from the request.
func PostText(w http.ResponseWriter, r *http.Request) {
	id, err := strconv.ParseInt(r.PathValue("id"), 10, 64)
	if err != nil {
		http.NotFound(w, r)
		return
	}
	p, err := GetPost(r.Context(), maago.DBFrom(r.Context()), id)
	if err != nil {
		http.NotFound(w, r)
		return
	}
	fmt.Fprintln(w, p.Title)
}

package blog

import (
	"fmt"
	"net/http"
	"net/http/httptest"
	"testing"
	"time"

	"github.com/maago-web/maago"
	"github.com/maago-web/maago/maagotest"
	_ "modernc.org/sqlite"

	"example.com/app/auth"
)

func TestViews(t *testing.T) {
	h, err := maago.Handler(maago.Config{DB: maagotest.DB(t), Apps: []maago.App{auth.App, App}, Auth: auth.Current, LoginURL: "/login"})
	if err != nil {
		t.Fatal(err)
	}
	for _, c := range []struct {
		method, path string
		code         int
		location     string
	}{
		{"GET", "/", 200, ""},
		{"GET", "/rows", 200, ""},
		{"GET", "/post/1", 404, ""},                            // no such row
		{"GET", "/post/abc", 404, ""},                          // not an int64
		{"GET", "/?page=abc", 400, ""},                         // not an int64
		{"GET", "/post/new", 303, "/login?next=%2Fpost%2Fnew"}, // u auth.User: sign in first
		{"POST", "/post/new", 401, ""},                         // a POST without a user
		{"GET", "/api/posts", 200, ""},
	} {
		r := httptest.NewRequest(c.method, c.path, nil)
		w := httptest.NewRecorder()
		h.ServeHTTP(w, r)
		if w.Code != c.code || w.Header().Get("Location") != c.location {
			t.Errorf("%s %s: %d %q, want %d %q", c.method, c.path, w.Code, w.Header().Get("Location"), c.code, c.location)
		}
	}
	if w := httptest.NewRecorder(); true {
		h.ServeHTTP(w, httptest.NewRequest("GET", "/api/posts", nil))
		if ct := w.Header().Get("Content-Type"); ct != "application/json" {
			t.Errorf("api_posts: %s", ct)
		}
	}
	_ = http.StatusOK
}

// Who gets each route: a row for each, a status for each role, or
// maago gen refuses the app, whose views take the user. A route added
// without its row fails here. A row that writes gets a fixture for each
// role that succeeds; maagotest.Skip leaves a role out of that row.
func TestPermissions(t *testing.T) {
	db := maagotest.DB(t)
	h, err := maago.Handler(maago.Config{DB: db, Apps: []maago.App{auth.App, App}, Auth: auth.Current, LoginURL: "/login"})
	if err != nil {
		t.Fatal(err)
	}
	now := time.Now()
	ann, err := auth.InsertUser(t.Context(), db, auth.NewUserInput{Email: "ann@example.com", VerifiedAt: &now})
	if err != nil {
		t.Fatal(err)
	}
	session, err := auth.SignIn(t.Context(), db, ann) // a session, as signing in starts
	if err != nil {
		t.Fatal(err)
	}
	p, err := InsertPost(t.Context(), db, NewPostInput{AuthorID: ann.ID, Title: "first"})
	if err != nil {
		t.Fatal(err)
	}
	want := func(nobody, ann int) map[string]int { return map[string]int{"nobody": nobody, "ann": ann} }
	maagotest.Permissions(t, []maago.App{App}, []maagotest.Role{{Name: "nobody", Handler: h}, {Name: "ann", Handler: maagotest.As(h, session)}},
		maagotest.Row{Route: "post_list", URL: PostListURL(), Want: want(200, 200)},
		maagotest.Row{Route: "post_rows", URL: PostRowsURL(), Want: want(200, 200)},
		maagotest.Row{Route: "post_detail", URL: PostDetailURL(p.ID), Want: want(200, 200)},
		maagotest.Row{Route: "post_new", URL: PostNewURL(), Want: want(303, 200)},
		maagotest.Row{Route: "post_create", URL: PostCreateURL(), Body: "title=second", Want: want(401, 303)},
		maagotest.Row{Route: "post_view", URL: PostViewURL(p.ID), Want: want(303, 303)},
		maagotest.Row{Route: "post_text", URL: PostTextURL(fmt.Sprint(p.ID)), Want: want(200, 200)},
		maagotest.Row{Route: "api_posts", URL: APIPostsURL(), Want: want(200, 200)},
	)
}
