// migrations are the project's migrations, in the binary: it migrates its
// database as it starts, where there is no maago command.
//
//go:embed migrations
var migrations embed.FS

func main() {
	ctx := context.Background()
	// The database's address is MAAGO_DATABASE_URL, or .env in the working
	// directory. Migrate first: what is applied already is skipped.
	applied, err := maago.Migrate(ctx, maago.DBConfig{}, migrations)
	if err != nil {
		log.Fatal(err)
	}
	for _, name := range applied {
		log.Printf("applied %s", name)
	}
	db, err := maago.OpenDB(ctx, maago.DBConfig{})
	if err != nil {
		log.Fatal(err)
	}
	// One store, for the site, the worker and the sweep alike.
	media := maago.LocalMedia{Dir: "media"}
	h, err := handler(db, media)
	if err != nil {
		log.Fatal(err)
	}
	// Handler checks a directory; with a bucket instead, this stops on an
	// empty field, a wrong key or no such bucket, not at the first file.
	if _, err := media.List(ctx, ""); err != nil {
		log.Fatal(err)
	}
	port := cmp.Or(os.Getenv("PORT"), "8000")
	log.Printf("listening on http://localhost:%s/", port)
	// The outbox's worker, beside the server.
	work, stop := context.WithCancel(ctx)
	worked := make(chan struct{})
	go func() {
		outbox.Run(work, db, deliver(db, media))
		close(worked)
	}()
	// The sweep, as the server starts and each day after: it deletes the
	// files no row keeps, a day old. The store is this database's alone.
	go func() {
		for {
			if err := maago.SweepMedia(maago.WithMedia(work, media), db, 0); err != nil {
				log.Printf("sweep: %v", err)
			}
			select {
			case <-work.Done():
				return
			case <-time.After(24 * time.Hour):
			}
		}
	}()
	// Serve stops on SIGINT or SIGTERM and lets running requests finish.
	// Then the worker takes no more jobs, and finishes the one in hand.
	err = maago.Serve(ctx, ":"+port, h)
	stop()
	<-worked
	if err != nil {
		log.Fatal(err)
	}
}

// deliver does the work of a job the outbox's worker hands it, a case for
// each Kind. An error, and the job goes again later: a kind it does not
// know stays, for the admin to show, rather than being dropped.
func deliver(db maago.DB, media maago.MediaStore) func(context.Context, outbox.Job) error {
	return func(ctx context.Context, j outbox.Job) error {
		ctx = maago.WithMedia(ctx, media) // the store Config.Media has
		switch j.Kind {
		case notes.NoteCreated:
			return keepText(ctx, db, j)
		}
		return fmt.Errorf("no job of kind %q", j.Kind)
	}
}

// keepText keeps a note's text as a file, and the note its key.
func keepText(ctx context.Context, db maago.DB, j outbox.Job) error {
	var created notes.Note
	if err := json.Unmarshal([]byte(j.Payload), &created); err != nil {
		return err
	}
	n, err := notes.GetNote(ctx, db, created.ID)
	if err != nil {
		return err
	}
	m, err := maago.SaveMediaFrom(ctx, fmt.Sprintf("note-%d.txt", n.ID), strings.NewReader(n.Text))
	if err != nil {
		return err
	}
	old := n.Export // a file of a job that went before, if any
	n.Export = m.Key
	if err := notes.UpdateNote(ctx, db, &n); err != nil {
		return errors.Join(err, maago.DeleteMedia(ctx, m.Key)) // no row points at it
	}
	if old != "" {
		return maago.DeleteMedia(ctx, old)
	}
	return nil
}

// handler puts the apps together, apart from main so that a test serves
// the same site.
func handler(db *sql.DB, media maago.MediaStore) (http.Handler, error) {
	seoul, err := time.LoadLocation("Asia/Seoul")
	if err != nil {
		return nil, err
	}
	return maago.Handler(maago.Config{
		DB:       db,
		Apps:     []maago.App{notes.App}, // each app's generated App
		TimeZone: seoul,                  // where the times people enter and see are
		Media:    media,
		Messages: maago.Messages{Required: "Write something."},
		OnError: func(ctx context.Context, r *http.Request, err error) {
			log.Printf("%s %s: %v", r.Method, r.URL.Path, err) // every 5xx
		},
	})
}

package main

import (
	"context"
	"io"
	"net/http"
	"net/http/httptest"
	"path/filepath"
	"strings"
	"testing"
	"time"

	"github.com/maago-web/maago"

	"example.com/app/notes"
	"example.com/app/outbox"
)

func TestSite(t *testing.T) {
	ctx := context.Background()
	cfg := maago.DBConfig{URL: "sqlite:" + filepath.Join(t.TempDir(), "app.db")}
	applied, err := maago.Migrate(ctx, cfg, migrations)
	if err != nil || len(applied) == 0 {
		t.Fatalf("migrate: %v %v", applied, err)
	}
	if again, err := maago.Migrate(ctx, cfg, migrations); err != nil || len(again) != 0 {
		t.Fatalf("migrate again: %v %v", again, err)
	}
	db, err := maago.OpenDB(ctx, cfg)
	if err != nil {
		t.Fatal(err)
	}
	media := maago.LocalMedia{Dir: t.TempDir()}
	h, err := handler(db, media)
	if err != nil {
		t.Fatal(err)
	}
	post := func(body string) *httptest.ResponseRecorder {
		r := httptest.NewRequest("POST", "/", strings.NewReader(body))
		r.Header.Set("Content-Type", "application/x-www-form-urlencoded")
		w := httptest.NewRecorder()
		h.ServeHTTP(w, r)
		return w
	}
	if w := post("text="); w.Code != 200 || !strings.Contains(w.Body.String(), "Write something.") {
		t.Errorf("empty: %d %s", w.Code, w.Body)
	}
	if w := post("text=hello"); w.Code != http.StatusSeeOther {
		t.Errorf("hello: %d %s", w.Code, w.Body)
	}
	// The note's job, one step of the worker: done, so none is left, not
	// even an hour later, when a failed one would go again.
	if found, err := outbox.RunOnce(ctx, db, time.Now(), deliver(db, media)); !found || err != nil {
		t.Errorf("worker: %v %v", found, err)
	}
	if found, err := outbox.RunOnce(ctx, db, time.Now().Add(time.Hour), deliver(db, media)); found || err != nil {
		t.Errorf("worker again: %v %v", found, err)
	}
	if err := deliver(db, media)(ctx, outbox.Job{Kind: "nope"}); err == nil {
		t.Error("a kind deliver does not know: no error")
	}
	// The worker kept the note's text as a file, which the sweep keeps:
	// a row has its key.
	all, err := notes.AllNotes(ctx, db)
	if err != nil || len(all) != 1 {
		t.Fatalf("notes: %v %v", all, err)
	}
	if err := maago.SweepMedia(maago.WithMedia(ctx, media), db, 0); err != nil {
		t.Errorf("sweep: %v", err)
	}
	f, err := maago.OpenMedia(maago.WithMedia(ctx, media), all[0].Export)
	if err != nil {
		t.Fatal(err)
	}
	defer f.Close()
	if b, _ := io.ReadAll(f); string(b) != "hello" {
		t.Errorf("the note's file: %q", b)
	}
	w := httptest.NewRecorder()
	h.ServeHTTP(w, httptest.NewRequest("GET", "/", nil))
	if !strings.Contains(w.Body.String(), "hello") || !strings.Contains(w.Body.String(), "KST") {
		t.Errorf("list: %s", w.Body)
	}
}

package notes

import (
	"context"

	"github.com/maago-web/maago"

	"example.com/app/outbox"
)

// NoteCreated is the Kind of the job a new note enqueues, which main's
// deliver goes by.
const NoteCreated = "note_created"

type NoteListIn struct {
	Form NoteForm `maago:"form"`
}

type NoteListPage struct {
	Notes []Note
	Form  NoteForm
}

func NoteList(ctx context.Context, db maago.DB, in NoteListIn) (NoteListPage, error) {
	notes, err := AllNotes(ctx, db)
	return NoteListPage{Notes: notes, Form: in.Form}, err
}

func CreateNote(ctx context.Context, tx maago.Tx, in NoteListIn) (maago.Result[NoteListPage], error) {
	if !in.Form.Valid() {
		notes, err := AllNotes(ctx, tx)
		return maago.Show(NoteListPage{Notes: notes, Form: in.Form}), err
	}
	n, err := InsertNote(ctx, tx, NewNoteInput{Text: in.Form.Text.Value})
	if err != nil {
		return maago.Result[NoteListPage]{}, err
	}
	// In the same transaction: with no note, no job.
	if _, err := outbox.Enqueue(ctx, tx, NoteCreated, n); err != nil {
		return maago.Result[NoteListPage]{}, err
	}
	return maago.SeeOther[NoteListPage](NoteListURL()), nil
}
