package main

import (
	"context"
	"flag"
	"log"
	"os"

	"github.com/maago-web/maago"
	authlib "github.com/maago-web/maago/contrib/auth"
	_ "modernc.org/sqlite"

	"example.com/app/auth"
	"example.com/app/site"
)

func main() {
	createSuperuser := flag.Bool("createsuperuser", false, "create a staff account from MAAGO_SUPERUSER_EMAIL and MAAGO_SUPERUSER_PASSWORD")
	flag.Parse()
	ctx := context.Background()
	db, err := maago.OpenDB(ctx, maago.DBConfig{})
	if err != nil {
		log.Fatal(err)
	}
	if *createSuperuser {
		u, err := auth.CreateSuperuser(ctx, db, os.Getenv("MAAGO_SUPERUSER_EMAIL"), os.Getenv("MAAGO_SUPERUSER_PASSWORD"))
		if err != nil {
			log.Fatal(err)
		}
		log.Printf("created superuser %s (id %d)", u.Email, u.ID)
		return
	}
	auth.SiteURL = "http://localhost:8000" // https://… in production: the cookie is then Secure
	authlib.Mail = authlib.ConsoleMail     // the mails go to the log
	h, err := maago.Handler(maago.Config{
		DB:       db,
		Apps:     []maago.App{auth.App, site.App},
		Auth:     auth.Current,
		LoginURL: "/login",
	})
	if err != nil {
		log.Fatal(err)
	}
	if err := maago.Serve(ctx, ":8000", h); err != nil {
		log.Fatal(err)
	}
}

package site

import (
	"context"

	"example.com/app/auth"
)

type MePage struct {
	Email string
}

// Me requires a sign-in: u auth.User.
func Me(ctx context.Context, u auth.User, in struct{}) (MePage, error) {
	return MePage{Email: u.Email}, nil
}

func APIMe(ctx context.Context, u auth.User, in struct{}) (MePage, error) {
	return MePage{Email: u.Email}, nil
}

// Staff is for staff only.
func Staff(ctx context.Context, u auth.User, in struct{}) (MePage, error) {
	if err := auth.RequireStaff(u); err != nil {
		return MePage{}, err
	}
	return MePage{Email: u.Email}, nil
}

package site

import (
	"context"
	"net/http"
	"net/http/httptest"
	"net/url"
	"strings"
	"testing"
	"time"

	"github.com/maago-web/maago"
	"github.com/maago-web/maago/maagotest"
	_ "modernc.org/sqlite"

	"example.com/app/auth"
)

func TestSignIn(t *testing.T) {
	db := maagotest.DB(t)
	if _, err := auth.CreateSuperuser(context.Background(), db, "me@example.com", "a long passphrase"); err != nil {
		t.Fatal(err)
	}
	auth.SiteURL = "http://localhost:8000"
	h, err := maago.Handler(maago.Config{DB: db, Apps: []maago.App{auth.App, App}, Auth: auth.Current, LoginURL: "/login"})
	if err != nil {
		t.Fatal(err)
	}
	do := func(r *http.Request, cookies ...*http.Cookie) *httptest.ResponseRecorder {
		for _, c := range cookies {
			r.AddCookie(c)
		}
		w := httptest.NewRecorder()
		h.ServeHTTP(w, r)
		return w
	}

	// Nobody signed in: a page goes to the login page, the API is a 401.
	if w := do(httptest.NewRequest("GET", "/me", nil)); w.Code != 303 || w.Header().Get("Location") != "/login?next=%2Fme" {
		t.Errorf("GET /me: %d %s", w.Code, w.Header().Get("Location"))
	}
	if w := do(httptest.NewRequest("GET", "/api/me", nil)); w.Code != 401 || w.Header().Get("WWW-Authenticate") != "Bearer" {
		t.Errorf("GET /api/me: %d %q", w.Code, w.Header().Get("WWW-Authenticate"))
	}

	// Sign in with email and password; the sign-in goes back to next.
	r := httptest.NewRequest("POST", "/login?next=%2Fme", strings.NewReader(url.Values{"email": {"me@example.com"}, "password": {"a long passphrase"}}.Encode()))
	r.Header.Set("Content-Type", "application/x-www-form-urlencoded")
	w := do(r)
	cookies := w.Result().Cookies()
	if w.Code != 303 || w.Header().Get("Location") != "/me" || len(cookies) != 1 || cookies[0].Secure {
		t.Fatalf("POST /login: %d %s %v", w.Code, w.Header().Get("Location"), cookies)
	}
	if w := do(httptest.NewRequest("GET", "/me", nil), cookies...); w.Code != 200 || !strings.Contains(w.Body.String(), "me@example.com") {
		t.Errorf("GET /me signed in: %d", w.Code)
	}
	if w := do(httptest.NewRequest("GET", "/staff", nil), cookies...); w.Code != 200 {
		t.Errorf("GET /staff as a superuser: %d", w.Code)
	}

	// A POST from another origin is refused: there is no CSRF token.
	r = httptest.NewRequest("POST", "/logout", nil)
	r.Header.Set("Origin", "https://elsewhere.example")
	if w := do(r, cookies...); w.Code != 403 {
		t.Errorf("cross-origin POST /logout: %d", w.Code)
	}
}

// Who gets each route. A test signs a user in with auth.SignIn, without
// the password; maagotest.As puts the session on every request.
func TestPermissions(t *testing.T) {
	db := maagotest.DB(t)
	h, err := maago.Handler(maago.Config{DB: db, Apps: []maago.App{auth.App, App}, Auth: auth.Current, LoginURL: "/login"})
	if err != nil {
		t.Fatal(err)
	}
	now := time.Now()
	ann, err := auth.InsertUser(t.Context(), db, auth.NewUserInput{Email: "ann@example.com", VerifiedAt: &now})
	if err != nil {
		t.Fatal(err)
	}
	boss, err := auth.CreateSuperuser(t.Context(), db, "boss@example.com", "a long passphrase")
	if err != nil {
		t.Fatal(err)
	}
	var roles []maagotest.Role
	for _, u := range []auth.User{ann, boss} {
		session, err := auth.SignIn(t.Context(), db, u)
		if err != nil {
			t.Fatal(err)
		}
		roles = append(roles, maagotest.Role{Name: u.Email, Handler: maagotest.As(h, session)})
	}
	roles = append([]maagotest.Role{{Name: "nobody", Handler: h}}, roles...)
	want := func(nobody, ann, staff int) map[string]int {
		return map[string]int{"nobody": nobody, "ann@example.com": ann, "boss@example.com": staff}
	}
	maagotest.Permissions(t, []maago.App{App}, roles,
		maagotest.Row{Route: "me", URL: MeURL(), Want: want(303, 200, 200)},
		maagotest.Row{Route: "api_me", URL: APIMeURL(), Want: want(401, 200, 200)},
		maagotest.Row{Route: "staff", URL: StaffURL(), Want: want(303, 403, 200)},
	)
}
