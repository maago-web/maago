package main

import (
	"context"
	"log"

	"github.com/maago-web/maago"
	_ "modernc.org/sqlite"

	"example.com/app/blog"
)

func main() {
	ctx := context.Background()
	db, err := maago.OpenDB(ctx, maago.DBConfig{})
	if err != nil {
		log.Fatal(err)
	}
	h, err := maago.Handler(maago.Config{DB: db, Apps: []maago.App{blog.App}, MediaDir: "media"}) // a form takes a file
	if err != nil {
		log.Fatal(err)
	}
	if err := maago.Serve(ctx, ":8000", h); err != nil {
		log.Fatal(err)
	}
}

package blog

import (
	"strings"
	"time"

	"github.com/maago-web/maago"
)

// PostForm takes Title, Status and Subtitle's rules from the model.
type PostForm struct {
	maago.Form `maago:"model=Post"`
	Title      maago.Field[string]
	Status     maago.Field[string]
	Subtitle   maago.Field[string]
}

// Validate checks across fields, after every field was read.
func (f *PostForm) Validate() {
	if f.Title.Error == "" && strings.EqualFold(f.Title.Value, f.Subtitle.Value) {
		f.Subtitle.Error = "Say something the title does not."
	}
}

// ReviewForm has no model: each field's rules are its own tags.
type ReviewForm struct {
	maago.Form
	Name    maago.Field[string]     `maago:"min=2,max=100,label='Your name',help='As it will be shown'"`
	Rating  maago.Field[int64]      `maago:"min_value=1,max_value=5"`
	Weight  maago.Field[float64]    `maago:"optional,min_value=0"`
	Count   maago.Field[int32]      `maago:"optional"`
	Kind    maago.Field[string]     `maago:"choices=praise/complaint"`
	Visited maago.Field[time.Time]  `maago:"optional"`
	Public  maago.Field[bool]       // a checkbox
	Photo   maago.Field[maago.File] `maago:"optional,accept=png/jpeg"`
}

-- blog/views.go --
package blog

import (
	"context"
	"errors"

	"github.com/maago-web/maago"
)

type NewPostIn struct {
	Form PostForm `maago:"form"`
}

type PostFormPage struct {
	Form PostForm
}

func NewPost(ctx context.Context, in NewPostIn) (PostFormPage, error) {
	return PostFormPage{Form: in.Form}, nil // not bound: no errors shown
}

func CreatePost(ctx context.Context, tx maago.Tx, in NewPostIn) (maago.Result[PostFormPage], error) {
	f := in.Form
	if !f.Valid() {
		return maago.Show(PostFormPage{Form: f}), nil
	}
	_, err := insert(ctx, tx, f)
	if f.Fail(err) { // the title is taken: the unique constraint said so
		return maago.Show(PostFormPage{Form: f}), nil
	}
	if err != nil {
		return maago.Result[PostFormPage]{}, err
	}
	return maago.SeeOther[PostFormPage](PostNewURL()), nil
}

// APICreatePost takes the same form as JSON; a wrong one is a 422.
func APICreatePost(ctx context.Context, tx maago.Tx, in NewPostIn) (Post, error) {
	f := in.Form
	p, err := insert(ctx, tx, f)
	var taken *maago.UniqueError
	if errors.As(err, &taken) {
		return Post{}, maago.Invalid(&f.Title, "A post has this title.")
	}
	return p, err
}

func insert(ctx context.Context, tx maago.Tx, f PostForm) (Post, error) {
	in := NewPostInput{Title: f.Title.Value, Status: maago.Set(f.Status.Value)}
	if f.Subtitle.Given {
		in.Subtitle = &f.Subtitle.Value
	}
	return InsertPost(ctx, tx, in)
}

type ReviewIn struct {
	Form ReviewForm `maago:"form"`
}

type ReviewPage struct {
	Form ReviewForm
	Sent bool
}

func Review(ctx context.Context, in ReviewIn) (ReviewPage, error) {
	return ReviewPage{Form: in.Form}, nil
}

// SendReview writes nothing to the database, yet a POST takes maago.Tx.
func SendReview(ctx context.Context, tx maago.Tx, in ReviewIn) (maago.Result[ReviewPage], error) {
	return maago.Show(ReviewPage{Form: in.Form, Sent: in.Form.Valid()}), nil
}

package blog

import (
	"net/http/httptest"
	"strings"
	"testing"

	"github.com/maago-web/maago"
	"github.com/maago-web/maago/maagotest"
	_ "modernc.org/sqlite"
)

func TestForms(t *testing.T) {
	h, err := maago.Handler(maago.Config{DB: maagotest.DB(t), Apps: []maago.App{App}, MediaDir: t.TempDir()})
	if err != nil {
		t.Fatal(err)
	}
	post := func(path, body, contentType string) *httptest.ResponseRecorder {
		r := httptest.NewRequest("POST", path, strings.NewReader(body))
		r.Header.Set("Content-Type", contentType)
		w := httptest.NewRecorder()
		h.ServeHTTP(w, r)
		return w
	}
	const form = "application/x-www-form-urlencoded"
	if w := post("/post/new", "title=&status=draft", form); w.Code != 200 || strings.Contains(w.Body.String(), "</label> <") {
		t.Errorf("no title: %d", w.Code) // required: drawn again with the error
	}
	if w := post("/post/new", "title=Hello&status=draft", form); w.Code != 303 {
		t.Errorf("a right form: %d", w.Code)
	}
	if w := post("/post/new", "title=Hello&status=draft", form); w.Code != 200 || !strings.Contains(w.Body.String(), "This value is already taken.") {
		t.Errorf("the same title: %d", w.Code)
	}
	if w := post("/api/posts", `{"title": "Hello", "status": "draft"}`, "application/json"); w.Code != 422 {
		t.Errorf("the same title, as JSON: %d", w.Code)
	}
	if w := post("/api/posts", `{"title": "Other", "status": "draft"}`, "application/json"); w.Code != 200 {
		t.Errorf("a right JSON form: %d %s", w.Code, w.Body)
	}
}
