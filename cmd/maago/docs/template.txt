package main

import (
	"context"
	"log"

	"github.com/maago-web/maago"
	_ "modernc.org/sqlite"

	"example.com/app/blog"
)

func main() {
	ctx := context.Background()
	db, err := maago.OpenDB(ctx, maago.DBConfig{})
	if err != nil {
		log.Fatal(err)
	}
	h, err := maago.Handler(maago.Config{DB: db, Apps: []maago.App{blog.App}})
	if err != nil {
		log.Fatal(err)
	}
	if err := maago.Serve(ctx, ":8000", h); err != nil {
		log.Fatal(err)
	}
}

package blog

import (
	"context"

	"github.com/maago-web/maago"
)

type PostListPage struct {
	Posts []Post
	Count int // computed here, not in the template
}

func PostList(ctx context.Context, db maago.DB, in struct{}) (PostListPage, error) {
	posts, err := AllPosts(ctx, db)
	return PostListPage{Posts: posts, Count: len(posts)}, err
}

type PostDetailIn struct {
	ID int64 `maago:"path=id"`
}

type PostDetailPage struct {
	Post Post
}

func PostDetail(ctx context.Context, db maago.DB, in PostDetailIn) (PostDetailPage, error) {
	p, err := GetPost(ctx, db, in.ID)
	return PostDetailPage{Post: p}, err
}

package blog

import (
	"context"
	"net/http/httptest"
	"regexp"
	"strings"
	"testing"
	"time"

	"github.com/maago-web/maago"
	"github.com/maago-web/maago/maagotest"
	_ "modernc.org/sqlite"
)

func TestPages(t *testing.T) {
	db := maagotest.DB(t)
	at := time.Date(2026, 10, 18, 9, 30, 0, 0, time.UTC)
	p, err := InsertPost(context.Background(), db, NewPostInput{Title: "Hello", PublishedAt: &at})
	if err != nil {
		t.Fatal(err)
	}
	h, err := maago.Handler(maago.Config{DB: db, Apps: []maago.App{App}})
	if err != nil {
		t.Fatal(err)
	}
	get := func(path string) string {
		w := httptest.NewRecorder()
		h.ServeHTTP(w, httptest.NewRequest("GET", path, nil))
		if w.Code != 200 {
			t.Fatalf("GET %s: %d", path, w.Code)
		}
		return w.Body.String()
	}
	list := get("/")
	for _, want := range []string{`href="` + PostDetailURL(p.ID) + `"`, "2026-10-18", "<title>Blog</title>"} {
		if !strings.Contains(list, want) {
			t.Errorf("no %s in:\n%s", want, list)
		}
	}
	css := regexp.MustCompile(`href="(/static/blog/site\.[0-9a-f]+\.css)"`).FindStringSubmatch(list)
	if css == nil {
		t.Fatalf("no hashed stylesheet in:\n%s", list)
	}
	get(css[1])
	if detail := get(PostDetailURL(p.ID)); !strings.Contains(detail, "<title>Hello</title>") || !strings.Contains(detail, "2026-10-18 09:30") {
		t.Errorf("detail:\n%s", detail)
	}
}
